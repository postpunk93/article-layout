```python
# tests/test_no_bare_control_modifier.py
"""Guard: keyboard presses must use ControlOrMeta (Cmd on macOS), never bare Control.

A bare press("Control+...") does not paste / select-all / bold on macOS, so it
would silently break every paste-based adapter there. ControlOrMeta resolves per-OS.
"""
from pathlib import Path

SRC = Path(__file__).resolve().parent.parent / "src"


def test_no_bare_control_press_in_adapters():
    offenders = []
    for py in SRC.glob("*_adapter.py"):
        for i, line in enumerate(py.read_text(encoding="utf-8").splitlines(), 1):
            # 'press("Control+' matches bare Control but NOT 'ControlOrMeta+'
            if 'press("Control+' in line:
                offenders.append(f"{py.name}:{i}")
    assert offenders == [], f"Use ControlOrMeta, not bare Control: {offenders}"
```

Run: `python -m pytest tests/test_no_bare_control_modifier.py -v`

**Step 3: Do the replacements**

**Step 3: Implement**

In `src/sessions.py` add `import sys` (near the other stdlib imports). Replace the `_CHROME_CANDIDATES = [...]` block with:

```python
def _chrome_candidates() -> list[Path]:
    """Likely paths to the installed Chrome for the current OS.

    Only gates "is Chrome installed" — open_context resolves Chrome itself via
    Playwright channel="chrome". Keep ALL OS knowledge here; adapters stay OS-blind.
    """
    if sys.platform == "darwin":
        return [
            Path("/Applications/Google Chrome.app/Contents/MacOS/Google Chrome"),
            Path.home() / "Applications/Google Chrome.app/Contents/MacOS/Google Chrome",
        ]
    if sys.platform.startswith("win"):
        return [
            Path(os.environ.get("PROGRAMFILES", r"C:\Program Files")) / "Google/Chrome/Application/chrome.exe",
            Path(os.environ.get("PROGRAMFILES(X86)", r"C:\Program Files (x86)")) / "Google/Chrome/Application/chrome.exe",
            Path(os.environ.get("LOCALAPPDATA", "")) / "Google/Chrome/Application/chrome.exe",
        ]
    return [Path("/usr/bin/google-chrome"), Path("/usr/bin/google-chrome-stable")]
```

Update `chrome_executable()` to iterate the function:

```python
    for path in _chrome_candidates():
        if str(path) and path.exists():
            return str(path)
    return None
```
